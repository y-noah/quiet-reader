cmake_minimum_required(VERSION 3.22.1)
project(quietreader_tokenizer LANGUAGES C CXX)
if(NOT EXISTS "${SENTENCEPIECE_SOURCE}/src/sentencepiece_processor.h")
    message(FATAL_ERROR "Prepare pinned SentencePiece source with scripts/models/prepare_native.py")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${SENTENCEPIECE_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
add_library(quietreader_sentencepiece SHARED sentencepiece_jni.cpp)
target_include_directories(quietreader_sentencepiece PRIVATE "${SENTENCEPIECE_SOURCE}/src")
target_link_libraries(quietreader_sentencepiece PRIVATE sentencepiece-static log)
target_link_options(quietreader_sentencepiece PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
